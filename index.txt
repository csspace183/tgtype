    function parseYandexMusicEmbed(text) {
      const trimmed = (text || '').trim();
      const iframeMatch = trimmed.match(/<iframe\b[^>]*\bsrc=["'](https?:\/\/[^"']+)["'][^>]*>/i);
      if (!iframeMatch) return null;

      const src = iframeMatch.at(1);
      const allowedHosts = /^(?:music\.yandex\.(?:ru|by|kz|uz|com)|(?:[\w-]+\.)?vk\.com|(?:[\w-]+\.)?vk\.ru|(?:[\w-]+\.)?vkvideo\.ru|(?:[\w-]+\.)?rutube\.ru|(?:[\w-]+\.)?dzen\.ru|(?:[\w-]+\.)?zvuk\.com|(?:[\w-]+\.)?yandex\.ru|(?:[\w-]+\.)?youtube\.com|(?:[\w-]+\.)?youtube-nocookie\.com)$/i;
      try {
        const urlObj = new URL(src);
        if (!allowedHosts.test(urlObj.hostname)) return null;

        const isTall = /\/iframe\/album\/\d+\/?(?:$|\?)/i.test(src) || /video|embed|map-widget/i.test(src);
        return {
          src: src,
          isAlbum: isTall
        };
      } catch (e) {
        return null;
      }
    }

    function sanitizePastedHtml(rawHtml) {
      const doc = new DOMParser().parseFromString(rawHtml, 'text/html');
      const allowedTags = new Set([
        'P', 'BR', 'B', 'STRONG', 'I', 'EM', 'U', 'S', 'STRIKE', 'DEL',
        'H1', 'H2', 'H3', 'H4', 'BLOCKQUOTE', 'UL', 'OL', 'LI', 'A', 'SPAN', 'DIV', 'FONT'
      ]);

      function cleanNode(node) {
        if (node.nodeType === 3) {
          return document.createTextNode(node.nodeValue);
        }
        if (node.nodeType !== 1) return null;

        const tag = node.tagName.toUpperCase();
        if (['STYLE', 'SCRIPT', 'META', 'LINK', 'XML', 'HEAD', 'TITLE', 'IFRAME', 'OBJECT'].includes(tag)) {
          return null;
        }

        const frag = document.createDocumentFragment();
        for (const child of Array.from(node.childNodes)) {
          const cleanedChild = cleanNode(child);
          if (cleanedChild) frag.appendChild(cleanedChild);
        }

        if (!allowedTags.has(tag)) {
          return frag;
        }

        let targetTag = tag.toLowerCase();
        if (['H1', 'H2', 'H4'].includes(tag)) targetTag = 'h3';
        if (tag === 'OL') targetTag = 'ul';
        if (tag === 'FONT') targetTag = 'span';

        const el = document.createElement(targetTag);

        if (['P', 'DIV', 'H1', 'H2', 'H3', 'H4', 'BLOCKQUOTE'].includes(tag)) {
          const align = (node.style && node.style.textAlign) || node.getAttribute('align');
          if (align && ['center', 'right', 'left', 'justify'].includes(align.toLowerCase())) {
            el.style.textAlign = align.toLowerCase();
          }
        }

        const textColor = (node.style && node.style.color) || node.getAttribute('color');
        if (textColor) {
          el.style.color = textColor;
        }

        if (tag === 'A') {
          const href = node.getAttribute('href');
          if (href && /^https?:\/\//i.test(href.trim())) {
            el.href = href.trim();
            el.target = '_blank';
          } else {
            return frag;
          }
        }

        el.appendChild(frag);
        return el;
      }

      const container = document.createElement('div');
      for (const child of Array.from(doc.body.childNodes)) {
        const cleaned = cleanNode(child);
        if (cleaned) container.appendChild(cleaned);
      }
      return container.innerHTML;
    }

    function compressAndInsertImage(file) {
      if (!file) return;
      const reader = new FileReader();
      reader.onload = event => {
        const img = new Image();
        img.onload = () => {
          const canvas = document.createElement('canvas');
          const MAX_WIDTH = 850;
          let width = img.width;
          let height = img.height;
          if (width > MAX_WIDTH) {
            height = Math.round(height * MAX_WIDTH / width);
            width = MAX_WIDTH;
          }
          canvas.width = width;
          canvas.height = height;
          const ctx = canvas.getContext('2d');
          ctx.fillStyle = '#fff';
          ctx.fillRect(0, 0, width, height);
          ctx.drawImage(img, 0, 0, width, height);

          const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
          storyEl.focus();
          document.execCommand('insertHTML', false, `<img src="${dataUrl}"><br>`);
          updatePublishState();
        };
        img.src = event.target.result;
      };
      reader.readAsDataURL(file);
    }

    function parseFirestoreRestDoc(fields) {
      if (!fields) return {};
      const readVal = v => {
        if (!v) return undefined;
        if ('stringValue' in v) return v.stringValue;
        if ('integerValue' in v) return Number(v.integerValue);
        if ('doubleValue' in v) return Number(v.doubleValue);
        if ('booleanValue' in v) return Boolean(v.booleanValue);
        if ('arrayValue' in v) return (v.arrayValue.values || []).map(readVal);
        if ('mapValue' in v) {
          const obj = {};
          const mapFields = v.mapValue.fields || {};
          for (const mk of Object.keys(mapFields)) {
            obj[mk] = readVal(mapFields[mk]);
          }
          return obj;
        }
        return undefined;
      };
      const out = {};
      for (const k of Object.keys(fields)) {
        out[k] = readVal(fields[k]);
      }
      return out;
    }

        return;
      }

      if (!hasUnsavedChanges()) return;

      const titleText = titleEl.innerText.trim();
      if (!titleText) {
        alert('Введите заголовок!');
        return;
      }

      actionBtn.disabled = true;
      actionBtn.classList.add('is-saving');
      setStatus('Публикация...');

      try {
        let targetSlug = currentSlug;
        if (!targetSlug) {
          let baseSlug = slugify(titleText);
          targetSlug = baseSlug;
          let counter = 1;
          while ((await db.collection('articles').doc(targetSlug).get()).exists) {
            targetSlug = `${baseSlug}${counter++}`;
          }
        }

        const now = Date.now();
        const payload = {
          title: titleEl.innerHTML,
          story: storyEl.innerHTML,
          updatedAt: now
        };

        if (isAdmin(currentUser)) {
          payload.customMeta = metaInfoEl.innerText.trim();
        } else {
          payload.customMeta = firebase.firestore.FieldValue.delete();
        }

        if (!currentArticleOwner) {
          payload.ownerUid = currentUser.uid;
          payload.ownerLogin = getLoginName(currentUser);
          currentArticleOwner = currentUser.uid;
        }

        await db.collection('articles').doc(targetSlug).set(payload, { merge: true });

        if (isAdmin(currentUser)) {
          metaInfoEl.textContent = payload.customMeta;
        } else {
          renderMeta(payload.ownerLogin || getLoginName(currentUser), now);
        }

        currentSlug = targetSlug;
        window.history.pushState({}, '', '/' + targetSlug);
        document.title = titleText;
        isViewMode = true;
        captureInitialSnapshot();
        setEditable(false);
        updateEditVisibility();
        updateViewsBadge();
        setStatus('');

        // Красивая заливка зелёным и надпись СОХРАНЕНО
        actionBtn.classList.remove('is-saving');
        actionBtn.disabled = false;
        actionBtn.textContent = 'СОХРАНЕНО';
        actionBtn.classList.add('is-saved');

        if (savedBtnTimer) clearTimeout(savedBtnTimer);
        savedBtnTimer = setTimeout(() => {
          savedBtnTimer = null;
          actionBtn.classList.remove('is-saved');
          actionBtn.textContent = 'РЕДАКТИРОВАТЬ';
          updatePublishState();
        }, 1800);
      } catch (e) {
        actionBtn.classList.remove('is-saving');
        setStatus('Ошибка публикации: ' + e.message, 'error');
        updatePublishState();
      }
    });
  </script>
</body>
